# =====================================================
# PlotFinder Backend — Python зависимости
# =====================================================
#
# Установка:
#   python3 -m venv venv
#   source venv/bin/activate     # Linux/macOS
#   venv\Scripts\activate.bat    # Windows
#   pip install -r requirements.txt
#
# Запуск:
#   uvicorn main:app --host 0.0.0.0 --port 8000 --reload
#
# Версии зафиксированы под боевое окружение PlotFinder MVP
# на сервере Selectel (на момент защиты ВКР, май 2026).
# Совместимость: Python 3.10+
#

# -----------------------------------------------------
# Веб-фреймворк и сервер
# -----------------------------------------------------

# FastAPI — основной веб-фреймворк
fastapi==0.116.1

# Uvicorn — ASGI-сервер для запуска FastAPI в production
# [standard] добавляет ускорители (uvloop, httptools, websockets, watchfiles)
uvicorn[standard]==0.35.0


# -----------------------------------------------------
# Валидация и сериализация данных
# -----------------------------------------------------

# Pydantic v2 — модели данных, валидация, JSON-сериализация
# Используется через BaseModel для request/response схем FastAPI
# pydantic-core ставится автоматически как зависимость pydantic
pydantic==2.11.7


# -----------------------------------------------------
# HTTP-клиент
# -----------------------------------------------------

# requests — HTTP-клиент для обращения к OpenRouter API (LLM-инференс Qwen3-32B)
requests==2.32.5


# -----------------------------------------------------
# Внешние интеграции
# -----------------------------------------------------

# NextGIS Toolbox SDK — официальный клиент для получения геометрии
# земельных участков из Росреестра по кадастровому номеру.
# Используется в endpoint /api/rosreestr2coord
# Документация: https://toolbox.nextgis.com/
toolbox-sdk==0.1.0b4


# -----------------------------------------------------
# Парсинг документов
# -----------------------------------------------------

# pdfplumber — извлечение текста и табличных данных из PDF.

pdfplumber==0.11.9
